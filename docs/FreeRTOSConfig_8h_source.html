42/* ── Cortex-M33 port requirements ────────────────────────────────────────────
43 * The RP2350_ARM_NTZ port (ARM Cortex-M33, Non-TrustZone) REQUIRES these
44 * three defines. Without them portmacrocommon.h throws a hard #error.
45 * NTZ = Non-TrustZone, so TRUSTZONE = 0.
46 * RP2350 has an FPU, enable it. MPU is not used in this project.
47 * ─────────────────────────────────────────────────────────────────────────── */
48#define configENABLE_FPU 1
49#define configRUN_FREERTOS_SECURE_ONLY 1 // added this to build on the secure vers of FreeRTOS
50#define configENABLE_MPU 0
51#define configENABLE_TRUSTZONE 0
52
53/* ── SMP — must be unconditional, NOT inside #if FREE_RTOS_KERNEL_SMP ────────
54 * The Raspberry Pi FreeRTOS-Kernel fork defines FREE_RTOS_KERNEL_SMP
55 * internally, but only AFTER it has already read this config file.
56 * If these are gated behind #if FREE_RTOS_KERNEL_SMP they are never seen
57 * and the port compiles in single-core mode — causing the
58 * "configUSE_CORE_AFFINITY is not supported in single core" error.
59 * ─────────────────────────────────────────────────────────────────────────── */
60#define configNUM_CORES 2
61#define configNUMBER_OF_CORES 2
62#define configTICK_CORE 0
63#define configRUN_MULTIPLE_PRIORITIES 1
64
65#if configNUM_CORES == 2
66#define configUSE_CORE_AFFINITY 1
67#else
68#define configUSE_CORE_AFFINITY 0
69#endif

72/* ── Cortex-M33 interrupt priorities ─────────────────────────────────────────
73 * RP2350 implements 3 priority bits → 8 levels (0 = highest, 7 = lowest).
74 * These are required by portasm.c (configMAX_SYSCALL_INTERRUPT_PRIORITY).
75 * ─────────────────────────────────────────────────────────────────────────── */
76#define configKERNEL_INTERRUPT_PRIORITY ( 7 << ( 8 - 3 ) ) /* 0xE0 */
77#define configMAX_SYSCALL_INTERRUPT_PRIORITY ( 5 << ( 8 - 3 ) ) /* 0xA0 */
78#define configMAX_API_CALL_INTERRUPT_PRIORITY configMAX_SYSCALL_INTERRUPT_PRIORITY
